function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error('Empty POST body received.');
    }
    const body = JSON.parse(e.postData.contents);
    const action = body.action || 'upsert_record';

    if (action === 'staff_login') {
      return handleStaffLogin_(body);
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const master = getOrCreateSheet_(ss, 'Master_Admissions', HEADERS);

    if (action === 'upload_file') {
      const folder = DriveApp.getFolderById(body.folderId);
      const commaIdx = body.dataUrl.indexOf(',');
      const base64 = commaIdx !== -1 ? body.dataUrl.slice(commaIdx + 1) : body.dataUrl;
      const mime = body.dataUrl.slice(5, body.dataUrl.indexOf(';'));
      const blob = Utilities.newBlob(Utilities.base64Decode(base64), mime, body.fileName);

      const existingFiles = folder.getFilesByName(body.fileName);
      while (existingFiles.hasNext()) {
        existingFiles.next().setTrashed(true);
      }
      const file = folder.createFile(blob);
      const fileUrl = file.getUrl();

      updateRowFileLink_(master, body.admissionId, body.docKey, fileUrl);

      let emailStatusMsg = 'SKIPPED';
      if (body.sendConfirmationEmail && body.studentEmail) {
        try {
          const pdfAttachment = blob.getAs(MimeType.PDF).setName(body.fileName);
          MailApp.sendEmail({
            to: String(body.studentEmail).trim(),
            subject: 'Admission Confirmation & Resume Copy - ' + body.admissionId + ' (' + body.studentName + ')',
            body: 'Dear ' + body.studentName + ',\n\nYour admission details for ' + body.course +
                  ' (' + body.branch + ') at Jain College of Engineering, Belagavi have been recorded.\n' +
                  'Admission ID: ' + body.admissionId + '\nDrive Folder: ' + folder.getUrl() +
                  '\n\nYour Official Student Admission Resume PDF is attached.',
            attachments: [pdfAttachment]
          });
          emailStatusMsg = 'SENT (' + body.studentEmail + ')';
        } catch (mailErr) {
          emailStatusMsg = 'EMAIL_ERROR: ' + mailErr.toString();
        }
        updateRowColumnByName_(master, body.admissionId, 'emailStatus', emailStatusMsg);
      }

      return jsonResponse_({ status: 'OK', fileUrl: fileUrl, emailStatus: emailStatusMsg });
    }

    const payload = body.record || body;
    const root = DriveApp.getFolderById(ROOT_FOLDER_ID);
    const yFolder = getOrCreateFolder_(root, 'Admissions2026');
    const cFolder = getOrCreateFolder_(yFolder, payload.course || 'GENERAL');
    const bFolder = getOrCreateFolder_(cFolder, payload.branch || 'GENERAL');
    const safeStudent = String(payload.candidateName10th || 'STUDENT').replace(/[^a-zA-Z0-9_ ]/g, '').trim().replace(/\s+/g, '_');
    const sFolder = getOrCreateFolder_(bFolder, safeStudent + '_' + payload.Admission_ID);

    payload.driveFolderPath = sFolder.getUrl();
    payload.emailStatus = payload.email ? 'SENT (' + payload.email + ')' : 'N/A';

    const data = master.getDataRange().getValues();
    let existingRowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(payload.Admission_ID)) {
        existingRowIndex = i + 1;
        break;
      }
    }

    const rowValues = HEADERS.map(h => payload[h] !== undefined ? payload[h] : '');

    if (existingRowIndex > 1) {
      const verifiedStaff = verifyStaffToken_(body.staffToken, ['ADMIN', 'ADMISSION_OFFICER']);
      if (!verifiedStaff) {
        return jsonResponse_({ status: 'ERROR', message: 'Unauthorized or expired staff session. Please sign in again.' });
      }
      master.getRange(existingRowIndex, 1, 1, HEADERS.length).setValues([rowValues]);
    } else {
      master.appendRow(rowValues);
    }

    const branchTabName = (String(payload.course) + '_' + String(payload.branch)).replace(/[^a-zA-Z0-9_]/g, '_').substring(0, 30);
    const branchSheet = getOrCreateSheet_(ss, branchTabName, HEADERS);
    if (existingRowIndex === -1) {
      branchSheet.appendRow(rowValues);
    }

    if (body.auditEntries && body.auditEntries.length > 0) {
      const auditSheet = getOrCreateSheet_(ss, 'Audit_Log', ['Timestamp', 'Admission_ID', 'Staff_User', 'Field', 'Old_Value', 'New_Value']);
      body.auditEntries.forEach(a => {
        auditSheet.appendRow([a.timestamp, a.admissionId, a.staffUser, a.field, a.oldVal, a.newVal]);
      });
    }

    return jsonResponse_({
      status: 'OK',
      folderId: sFolder.getId(),
      driveUrl: sFolder.getUrl()
    });
  } catch (err) {
    return jsonResponse_({ status: 'ERROR', message: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

function doGet() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName('Master_Admissions');
    if (!sheet || sheet.getLastRow() < 2) {
      return jsonResponse_({ status: 'OK', records: [] });
    }
    const values = sheet.getDataRange().getValues();
    const headers = values[0];
    const records = values.slice(1).map(row => {
      const obj = {};
      headers.forEach((h, idx) => { obj[h] = row[idx]; });
      return obj;
    });
    return jsonResponse_({ status: 'OK', records: records });
  } catch (err) {
    return jsonResponse_({ status: 'ERROR', message: err.toString() });
  }
}

function getSessionSecret_() {
  const props = PropertiesService.getScriptProperties();
  let secret = props.getProperty('SESSION_SECRET');
  if (!secret) {
    secret = Utilities.getUuid() + '-' + Utilities.getUuid();
    props.setProperty('SESSION_SECRET', secret);
  }
  return secret;
}

function sha256Hex_(input) {
  const rawHash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input, Utilities.Charset.UTF_8);
  return rawHash.map(b => ('0' + (b & 0xFF).toString(16)).slice(-2)).join('');
}

function setupStaffUsersSheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = getOrCreateSheet_(ss, 'Staff_Users', ['Username', 'Password_SHA256', 'Role', 'Active']);
  if (sheet.getLastRow() < 2) {
    sheet.appendRow(['admin_jce', sha256Hex_('Admin@JCE2026'), 'ADMIN', 'YES']);
    sheet.appendRow(['officer_jce', sha256Hex_('Officer@JCE2026'), 'ADMISSION_OFFICER', 'YES']);
    sheet.appendRow(['viewer_jce', sha256Hex_('Viewer@JCE2026'), 'VIEWER', 'YES']);
  }
}

function handleStaffLogin_(body) {
  const username = String(body.username || '').trim();
  const password = String(body.password || '');
  if (!username || !password) {
    return jsonResponse_({ status: 'ERROR', message: 'Username and password are required.' });
  }
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('Staff_Users');
  if (!sheet) {
    return jsonResponse_({ status: 'ERROR', message: 'Staff_Users table not initialized. Run setupStaffUsersSheet first.' });
  }
  const inputHash = sha256Hex_(password);
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    const [rowUser, rowHash, rowRole, rowActive] = rows[i];
    if (String(rowUser).trim().toLowerCase() === username.toLowerCase()) {
      if (String(rowActive).toUpperCase() !== 'YES') {
        return jsonResponse_({ status: 'ERROR', message: 'Account is disabled. Contact Administrator.' });
      }
      if (String(rowHash).trim() === inputHash) {
        const expiresAt = Date.now() + (8 * 60 * 60 * 1000);
        const payloadStr = `${rowUser}|${rowRole}|${expiresAt}`;
        const signature = sha256Hex_(payloadStr + '|' + getSessionSecret_());
        const token = Utilities.base64Encode(`${payloadStr}|${signature}`);
        return jsonResponse_({
          status: 'OK',
          username: rowUser,
          role: rowRole,
          token: token,
          expiresAt: expiresAt
        });
      }
    }
  }
  return jsonResponse_({ status: 'ERROR', message: 'Invalid username or passcode.' });
}

function verifyStaffToken_(token, allowedRoles) {
  if (!token) return null;
  try {
    const decoded = Utilities.newBlob(Utilities.base64Decode(token)).getDataAsString();
    const parts = decoded.split('|');
    if (parts.length !== 4) return null;
    const [username, role, expiresAtStr, signature] = parts;
    if (Date.now() > Number(expiresAtStr)) return null;
    const expectedSig = sha256Hex_(`${username}|${role}|${expiresAtStr}|${getSessionSecret_()}`);
    if (signature !== expectedSig) return null;
    if (allowedRoles && allowedRoles.indexOf(role) === -1) return null;
    return { username, role };
  } catch (e) {
    return null;
  }
}

function updateRowFileLink_(sheet, admissionId, docKey, fileUrl) {
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const resumeCol = headers.indexOf('resumePdfDriveUrl') + 1;
  const linksCol = headers.indexOf('docLinksJson') + 1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(admissionId)) {
      if (docKey === 'generatedResumePdf' && resumeCol > 0) {
        sheet.getRange(i + 1, resumeCol).setValue(fileUrl);
      } else if (linksCol > 0) {
        let map = {};
        try { map = JSON.parse(data[i][linksCol - 1] || '{}'); } catch (e) {}
        map[docKey] = fileUrl;
        sheet.getRange(i + 1, linksCol).setValue(JSON.stringify(map));
      }
      break;
    }
  }
}

function updateRowColumnByName_(sheet, admissionId, colName, value) {
  const data = sheet.getDataRange().getValues();
  const colIdx = data[0].indexOf(colName) + 1;
  if (colIdx < 1) return;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(admissionId)) {
      sheet.getRange(i + 1, colIdx).setValue(value);
      break;
    }
  }
}

function jsonResponse_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateFolder_(parent, name) {
  const clean = String(name).replace(/[/\\?%*:|"<>]/g, '-').trim();
  const it = parent.getFoldersByName(clean);
  return it.hasNext() ? it.next() : parent.createFolder(clean);
}

function getOrCreateSheet_(ss, name, headers) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
  }
  return sheet;
}
